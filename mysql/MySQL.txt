Q.출판사별 출판사의 평균 도서 가격보다 비싼 도서를 조회

select b1.bookname from book b1
where b1. price > (
select avg(b2.price)
from book b2
where b2.publisher=b1.publisher);

select b1.bookname
from book b1 join (
select publisher, avg(price) as avg_price
from book
group by publisher ) b2
on b1.publisher = b2.publisher
where b1.price > b2.avg_price;

select *
from book b1 join (
select publisher, avg(price) as avg_price
from book
group by publisher ) b2
on b1.publisher = b2.publisher
where b1.price > b2.avg_price;
